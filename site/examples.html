Examples

Every snippet below is real Vect that runs on vectc. Run with vectc file.vt. Programs needing input (marked) take typed lines on stdin — pipe them or type interactively.

1 · Variables & math
vectfn main
    a10
    b20
    (c.a+b)              cmt c = 30
    echo "Sum:"
    echo c
    val16
    (root.sqrt.val)      cmt root = 4
    echo root
    (shlRes.bit shl a, 2)   cmt 10 << 2 = 40
    echo shlRes
    (shrRes.bit shr b, 1)   cmt 20 >> 1 = 10
    echo shrRes
main()

16 · Persistent todo (KV store)
vectfn main
    kvopen db, "todo.kv"
    (n.kvexists db, "count")
    if n == 0
        kvset db, "count", 0
    echo "1 add, 2 list:"
    in(choice)
    if choice == 1
        (c.kvget db, "count")
        in(item)
        (key.cat "t", c)
        kvset db, key, item
        (c.c+1)
        kvset db, "count", c
        kvsave db
    els
        (c.kvget db, "count")
        i0
        lo100w i < c
            (key.cat "t", i)
            (v.kvget db, key)
            echo v
            (i.i+1)
    kvclose db
main()

18 · String reverse
vectfn main
    in(s)
    (clean.cat s, "")
    (n.len clean)
    out""
    (i.n-1)
    lo100w i >= 0
        (code.char clean, i)
        (ch.chr code)
        (out.cat out, ch)
        (i.i-1)
    echo out
main()
19 · Tic-tac-toe (2 player)
vectfn line3.a, b, c
    if a == 0
        rtn 0
    if a == b
        if b == c
            rtn 1
    rtn 0
vectfn main
    b(9)
    turn1
    moves0
    done0
    lo9w done == 0
        in(cell)
        (old.get b, cell)
        if old == 0
            set b, cell, turn
            (moves.moves+1)
            cmt check all 8 lines, e.g. top row:
            (x.get b, 0)
            (y.get b, 1)
            (z.get b, 2)
            (w.line3 x, y, z)
            if w == 1
                echo turn
                done1
            if moves == 9
                echo "draw"
                done1
            if turn == 1
                turn2
            els
                turn1
        els
            echo "taken"
main()
20 · Countdown timer
vectfn main
    in(s)
    (ms.s*1000)
    (t.clock)
    (end.t+ms)
    done0
    lo1000w done == 0
        (now.clock)
        (left.end-now)
        (sec.left/1000)
        echow sec, 200
        if left <= 0
            done1
    echo "done"
main()
21 · Recursion (factorial)
vectfn fact.n
    if n == 0
        rtn 1
    (m.n-1)
    (r.fact m)
    (p.r*n)
    rtn p
vectfn main
    (f.fact 5)
    echo f          cmt 120
main()
